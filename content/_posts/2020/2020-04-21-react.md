## javascript

```javascript
function HelloWorld() {
  [counter, setCounter] = React.useState(1);

  const increment = () => {
    setCounter((oldVal) => oldVal + 1);
    console.log("counter was", counter);
  };

  return (
    <div>
      {counter}
      <button onClick={increment}>increment</button>
    </div>
  );
}

ReactDOM.render(<HelloWorld />, document.querySelector("#app"));
```
