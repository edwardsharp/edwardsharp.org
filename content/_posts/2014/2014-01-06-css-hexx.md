  content: " ";
  width: 0;
  height: 0;
  border-bottom: 30px solid #ff6600;
  border-left: 52px solid transparent;
  border-right: 52px solid transparent;
  position: absolute;
  top: -30px;
}

.hex {
  margin-top: 30px;
  width: 102px;
  height: 66px;
  background-color: #ff6600;
  position: relative;
}

.hex:after {
  content: "";
  width: 0;
  position: absolute;
  bottom: -30px;
  border-top: 30px solid #ff6600;
  border-left: 52px solid transparent;
  border-right: 52px solid transparent;
}

.bump {
  margin: 66px;
}
