## css

```css

#sortable2,
#sortable3 {
  list-style-type: none;
  margin: 0;
  float: left;
  margin-right: 10px;
  background: #eee;
  padding: 5px;
  width: 143px;
}

#sortable1 li,
#sortable2 li,
#sortable3 li {
  margin: 5px;

  font-size: 1.2em;
  width: 120px;
}
```

## javascript

```javascript
$(function () {
  $("ul#sortable1").sortable({});

  $("ul#sortable2").sortable({});

  $("ul#sortable3").sortable({});
  $("#sortable1, #sortable2, #sortable3").disableSelection();
});
```
